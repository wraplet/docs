import {
    AbstractWraplet,
    AbstractDependentWraplet,
    DDM,
    type WrapletDependencyMap,
} from "wraplet";

// Small leaf wraplets - each one owns a single, real DOM node
// and exposes a tiny, typed API around it.

class NameInput extends AbstractWraplet<HTMLInputElement> {
    public getValue(): string {
        return this.node.value;
    }
}

class SubmitButton extends AbstractWraplet<HTMLButtonElement> {
    public onClick(callback: () => void): void {
        // Lifecycle-aware listener: removed automatically on destroy.
        this.nodeManager.addListener("click", () => callback());
    }
}

class Output extends AbstractWraplet<HTMLElement> {
    public setText(value: string): void {
        this.node.textContent = value;
    }
}

// Declarative, typed dependency map.
// `satisfies WrapletDependencyMap` keeps full inference - hover any
// property in your IDE and you'll see the exact wraplet class behind it.
const map = {
    nameInput: {
        selector: "[data-js-greeter__name]",
        Class: NameInput,
        required: true,
        multiple: false,
    },
    submit: {
        selector: "[data-js-greeter__submit]",
        Class: SubmitButton,
        required: true,
        multiple: false,
    },
    output: {
        selector: "[data-js-greeter__output]",
        Class: Output,
        required: true,
        multiple: false,
    },
} satisfies WrapletDependencyMap;

class Greeter extends AbstractDependentWraplet<HTMLElement, typeof map> {
    protected override async onInitialize() {
        // `this.d` is fully typed from the map above.
        // Try renaming `submit` to `submitButton` - TypeScript will
        // immediately flag every place that needs updating.
        this.d.submit.onClick(() => {
            const name = this.d.nameInput.getValue() || "stranger";
            this.d.output.setText(`Hello, ${name}!`);
        });
    }
}

const element: HTMLElement = document.querySelector("[data-js-greeter]");

// DDM manages dependencies. The instantiation/initialization
// can be simplified with helpers.
const greeter = new Greeter(new DDM(element, map));
await greeter.wraplet.initialize();
