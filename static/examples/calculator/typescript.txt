import { AbstractWraplet, AbstractDependentWraplet, type WrapletDependencyMap } from "wraplet";

// Follow the numbered comments.

/**
 * 1. Dependencies.
 */

class Number extends AbstractWraplet<HTMLInputElement> {

    public getValue(): number {
        return parseInt(this.node.value);
    }

    protected override supportedNodeTypes() {
        return this.supportedNodeTypesGuard([HTMLInputElement]);
    }

}

class Result extends AbstractWraplet<HTMLInputElement> {

    public setValue(value: number): void {
        this.node.value = value.toString();
    }

    protected override supportedNodeTypes() {
        return this.supportedNodeTypesGuard([HTMLInputElement]);
    }

}

class Calculate extends AbstractWraplet {

    public addClickListener(listener: () => void) {
        this.nodeManager.addListener("click", (e: Event) => {
            listener();
        });
    }

}

/**
 * 2. Dependency Map.
 */

const map = {

    numbers: {
        selector: "[data-js-calculator__number]",
        Class: Number,
        required: true,
        multiple: true,
    },

    result: {
        selector: "[data-js-calculator__result]",
        Class: Result,
        required: true,
        multiple: false,
    },

    calculate: {
        selector: "[data-js-calculator__calculate]",
        Class: Calculate,
        required: true,
        multiple: false,
    },

} satisfies WrapletDependencyMap;

/**
 * 3. Implementation.
 */

class Calculator extends AbstractDependentWraplet<HTMLElement, typeof map> {

    protected override async onInitialize() {

        this.d.calculate.addClickListener(() => {
            let sum = 0;

            for (const nmbr of this.d.numbers) {
                sum += nmbr.getValue();
            }

            this.d.result.setValue(sum);
        });

    }

    public static async create(parentNode: ParentNode) {
        return this.createAndInitializeDependentWraplets(parentNode, "data-js-calculator", map);
    }

}

/**
 * 4. Initialization.
 */

const calculators = await Calculator.create(document);
